# Question 30
a = 10
b = 10.0
c = "10"
d = True
e = "True"
f = None
g = "None"

print(a, "->", type(a))
print(b, "->", type(b))
print(c, "->", type(c))
print(d, "->", type(d))
print(e, "->", type(e))
print(f, "->", type(f))
print(g, "->", type(g))

# Identified types:
# 10     -> int
# 10.0   -> float
# "10"   -> str
# True   -> bool
# "True" -> str
# None   -> NoneType
# "None" -> str
